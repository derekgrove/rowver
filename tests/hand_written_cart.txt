"""A hand-written recipe shaped like the LPC study's: its fileset and helpers live beside it.

The real one is not in this public repository; this keeps its shape: the sys.path line,
FILESET imported from fileset.py, helpers imported as H, and columns such as H.ratio.
"""
import sys
from pathlib import Path

import awkward as ak
import numpy as np

from rowver import Cart

sys.path.insert(0, str(Path(__file__).resolve().parent))
from fileset import FILESET
import helpers as H


def keep(events):
    """Exactly two leptons, both of the best quality."""
    return (events.Nlep == 2) & (ak.sum(events.LepQual_lep == 0, axis=1) == 2)


CART = Cart(
    fileset=FILESET,
    keep=keep,
    columns={
        "weight": "weight",
        "Nlep": "Nlep",
        "Ngold": lambda events: ak.sum(events.LepQual_lep == 0, axis=1),
        "MET": "MET",
        "lep_sign_flav": H.lepton_category,
        "pass_eta_lep": lambda events: ak.any(np.abs(events.Eta_lep) < 1.4442, axis=1),
        "lepE_ratio": H.ratio,
    },
    labels={
        "lep_sign_flav": {
            0: "OS_ee", 1: "OS_emu", 2: "OS_mumu",
            3: "SS_ee", 4: "SS_emu", 5: "SS_mumu",
        },
    },
    title="Two gold leptons",
    note="A stand-in with the shape of a hand-written study.",
)
